<script>
const slides = document.querySelectorAll('.slide');
const indicator = document.getElementById('page-indicator');
let currentIndex = 0;
function updateUI() {
slides.forEach((slide, index) => {
if (index === currentIndex) {
slide.classList.add('active');
} else {
slide.classList.remove('active');
}
});
indicator.innerText = `${currentIndex + 1} / ${slides.length}`;
// バッテリーのアニメーション発火 (12枚目 = index 11)
if(currentIndex === 11) {
// アニメーションを確実に「増えていく」挙動にするため一度0%に戻す
document.getElementById('bar-old').style.width = '0%';
document.getElementById('bar-new').style.width = '0%';
setTimeout(() => {
document.getElementById('bar-old').style.width = '45%';
document.getElementById('bar-new').style.width = '100%';
}, 50);
} else {
document.getElementById('bar-old').style.width = '0%';
document.getElementById('bar-new').style.width = '0%';
}
}
function nextSlide() {
if (currentIndex < slides.length - 1) {
currentIndex++;
updateUI();
}
}
function prevSlide() {
if (currentIndex > 0) {
currentIndex--;
updateUI();
}
}
function goToSlide(index) {
currentIndex = index;
updateUI();
}
// 初期化
updateUI();
</script>
